const linhas = {};
const lerLinha = (f, n) => (linhas[f] = linhas[f] || fs.readFileSync(path.join(process.cwd(), f), 'utf8').split('\n'))[n - 1];
const CTX = (() => { try { return fs.readFileSync(process.env.NOTA_CTX_A, 'utf8'); } catch { return null; } })();
const semRegra = [];
const out = [];
for (const o of V.ocorrencias) {
  const regras = F[o.arquivo];
  let r = null;
  if (regras) r = regras.find(([de, ate]) => o.linha >= de && o.linha <= ate);
  if (!r) { semRegra.push(`${o.arquivo}:${o.linha}:${o.id}`); continue; }
  let [, , obj, papel, tr, nota] = r;
  // O OBJETO sai do identificador dentro da família da regra: `PARECER:` e `ANOTACAO:` seguem o identificador lido.

  else if (obj.startsWith('ANOTACAO:')) obj = 'ANOTACAO:' + o.id;
  // refinamento por texto: um comentário de linha inteira é "comentário", qualquer que seja a faixa
  const texto = lerLinha(o.arquivo, o.linha) || '';
  let refinado = false;
  if (eComentario(texto) && !papel.startsWith('teste') && papel !== 'comentário' && papel !== 'conteúdo RAG' && !String(papel).startsWith('script')) { papel = 'comentário'; refinado = true; }
  if (eComentario(texto) && papel.startsWith('teste') && papel !== 'teste: título') { papel = 'teste: comentário'; refinado = true; }
  if (o.arquivo.startsWith('lib/__tests__/') && /^\s*(test|it|describe)(\.\w+)?\(/.test(texto) && papel.startsWith('teste')) { papel = 'teste: título'; refinado = true; }
  // ARTIGOS: o transporte 2 é MEDIDO, não suposto: o valor entre aspas da linha aparece (ou não) no contexto capturado (elegivel-auto-A)?
  let chegaAoContexto;
  if (obj === 'ARTIGOS:score' && CTX) {
    // ⚠ Um valor curto e sem espaço (por exemplo, `unit: "score"`) está contido em QUALQUER contexto, e a inclusão não o
    //   discrimina: o campo `unit` é impresso por knowledge.ts:285 como `- **<valor> <unidade>** (<operador>): ...`, e é
    //   essa forma que se procura. Para os demais, o valor entre aspas precisa ter espaço e 10 caracteres ou mais.
    const valores = [...texto.matchAll(/"([^"]*score[^"]*)"/g)].map((m) => m[1]);
    if (/^\s*unit:\s*"score"/.test(texto)) chegaAoContexto = /\*\*[^*\n]*\sscore\*\* \(/.test(CTX);
    else if (valores.length && valores.every((v) => v.includes(' ') && v.length >= 10)) chegaAoContexto = valores.every((v) => CTX.includes(v));
    else chegaAoContexto = null;
    tr = chegaAoContexto === true ? 2 : chegaAoContexto === false ? 'não chega (nesta captura)' : '? (instrumento não decide)';
  }
  out.push({ arquivo: o.arquivo, linha: o.linha, id: o.id, objeto: obj, papel, transporte: tr, chegaAoContextoNaCapturaElegivelA: chegaAoContexto ?? null, nota, refinadoPorComentario: refinado, texto: texto.trim().slice(0, 200) });
}
if (semRegra.length) { console.error('SEM REGRA:', semRegra.length, semRegra.slice(0, 40).join(' ')); process.exit(1); }

const chave = (o) => `${o.objeto} | ${o.papel} | T${o.transporte}`;
const grupos = {};
for (const o of out) { const g = (grupos[chave(o)] = grupos[chave(o)] || { n: 0, arquivos: new Set() }); g.n++; g.arquivos.add(o.arquivo); }
const resumo = Object.entries(grupos).map(([k, g]) => ({ chave: k, ocorrencias: g.n, arquivos: g.arquivos.size })).sort((a, b) => a.chave.localeCompare(b.chave));
const porObjeto = {};
for (const o of out) { porObjeto[o.objeto] = porObjeto[o.objeto] || { total: 0, producao: 0, teste: 0 }; porObjeto[o.objeto].total++; if (o.arquivo.startsWith('lib/__tests__/')) porObjeto[o.objeto].teste++; else porObjeto[o.objeto].producao++; }
