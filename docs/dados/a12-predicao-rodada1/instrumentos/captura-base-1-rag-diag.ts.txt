const DIMS = 1024;

type ChunkSimulado = { id: string; score: number; metadata: Record<string, unknown> };
type Desfecho = ChunkSimulado[] | 'lanca';
function chunk(id: string, artigo: string, score: number): ChunkSimulado {
  return { id, score, metadata: { article_id: artigo, article_year: 2020, article_type: 'paper', chunk_type: 'claim', chunk_index: 1, text: 'texto do ' + id, verbatim_quote: 'verbatim do ' + id, page: 10, locator_type: null, locator_id: 'loc-' + id, usable_as: null } };
}
function payloadRag() {
  const balde = { total: 4, valid: 4, warning: 0, critical: 0, avgCR: 0.03 };
  return {
    projectName: 'Projeto de ensaio A.30',
    alternatives: [{ code: 'A1', name: 'Alternativa 1' }, { code: 'A2', name: 'Alternativa 2' }],
    finalScores: [{ code: 'A1', name: 'Alternativa 1', score: 0.62 }, { code: 'A2', name: 'Alternativa 2', score: 0.38 }],
    individualStats: { Benefits: balde, Opportunities: balde, Costs: balde, Risks: balde },
    bocrWeights: { Benefits: 0.37, Opportunities: 0.15, Costs: 0.2, Risks: 0.28 },
    bocrConsistency: { cr: 0.0106, lambda: 4.03 },
    overallStats: { total: 4, valid: 4, warning: 0, critical: 0 },
  };
}
async function executarRag(habilitado: string | undefined, embedFalha: boolean, porConsulta: Desfecho[]) {
  jest.resetModules();
  for (const [nome, valor] of Object.entries(CHAVES_FALSAS_RAG)) process.env[nome] = valor;
  if (habilitado === undefined) delete process.env.USE_RAG_SEMANTIC; else process.env.USE_RAG_SEMANTIC = habilitado;
  let consultasAoIndice = 0;
  const system: string[] = [];
  const messages: Array<Array<{ role: string; content: string }>> = [];
  jest.doMock('voyageai', () => ({ VoyageAIClient: class { async embed() { if (embedFalha) throw new Error('voyage indisponivel no ensaio'); return { data: [{ embedding: new Array(DIMS).fill(0.1) }] }; } } }));
  jest.doMock('@upstash/vector', () => ({ Index: class { async query() { const desfecho = porConsulta[consultasAoIndice]; consultasAoIndice += 1; if (desfecho === 'lanca') throw new Error('indice indisponivel no ensaio'); return desfecho ?? []; } } }));
  jest.doMock('@anthropic-ai/sdk', () => ({ __esModule: true, default: class { messages = { create: async (args: any) => { system.push(args.system); messages.push(args.messages); return { content: [{ type: 'text', text: 'PARECER DE ENSAIO' }] }; } }; } }));
  const spies = ['warn', 'log', 'error'].map((m) => jest.spyOn(console, m as 'log').mockImplementation(() => undefined));
  try {
    const { POST } = require('@/app/api/ai-reviewer/route');
    const res = await POST({ json: async () => payloadRag() } as any);
    await res.json();
    return { system: system[0] ?? '', messages: messages[0] ?? [] };
  } finally { spies.forEach((s) => s.mockRestore()); }
}
const cincoChunks: ChunkSimulado[] = [chunk('k1', 'artigo_alfa', 0.95), chunk('k2', 'artigo_alfa', 0.94), chunk('k3', 'artigo_alfa', 0.93), chunk('k4', 'artigo_alfa', 0.92), chunk('k5', 'artigo_beta', 0.91)];

test('captura da base: rag-semantic-states, as quatro condições', async () => {
  const casos: Record<string, Desfecho[]> = {
    comResultados: Array(5).fill(cincoChunks), vazio: Array(5).fill([]), erro: Array(5).fill('lanca'), misto: [cincoChunks, [], 'lanca', [], 'lanca'],
  };
  const resumo: any = {};
  for (const [nome, desfechos] of Object.entries(casos)) {
    const r = await executarRag('true', false, desfechos);
    const json = JSON.stringify(r.messages);
    fs.writeFileSync(path.join(SAIDA, `rag-${nome}-messages.json`), json, 'utf8');
    fs.writeFileSync(path.join(SAIDA, `rag-${nome}-conteudo.txt`), r.messages.map((m) => m.content).join('\n'), 'utf8');
    resumo[nome] = { nMessages: r.messages.length, sha256System: sha(r.system), sha256Messages: sha(json), bytesJson: Buffer.byteLength(json, 'utf8') };
  }
  fs.writeFileSync(path.join(SAIDA, 'rag-resumo.json'), JSON.stringify(resumo, null, 1));
});
