/** Um elemento como o produtor de qualidade o devolve (`app/api/response-quality/route.ts:377-402`). */
function itemDoProdutor(id: string, cr: number | undefined) {
  const status =
    cr === undefined || cr === 0 ? 'DESCONHECIDO' : cr <= 0.1 ? 'CONFIÁVEL' : cr <= 0.15 ? 'REVISAR' : cr <= 0.2 ? 'SUSPEITO' : 'CRÍTICO';
  const semCR = cr === undefined;
  return {
    respondentId: id,
    isSimulated: false,
    status,
    overallScore: 50,
    ...(semCR ? {} : { cr }),
    metrics: semCR ? {} : { avgCR: cr, crBOCR: cr, crBenefits: cr, crOpportunities: cr, crCosts: cr, crRisks: cr },
    flags: [],
    recommendation: '',
  };
}

const balde = (total: number, valid: number, warning = 0, critical = 0) => ({ total, valid, warning, critical, avgCR: 0.05 });

/** COPIA LITERAL da redução de `page.tsx:1058-1088`. A tela não é executada por teste algum. */
function individualStatsDaTela(respondentesEnviados: any[]) {
  let totalCR = 0;
  const acc: any = respondentesEnviados.reduce(
    (acc: any, r: any) => {
      let crValue = 0;
      if (typeof r.cr === 'number') crValue = r.cr;
      else if (r.metrics && typeof r.metrics.avgCR === 'number') crValue = r.metrics.avgCR;
      else if (r.consistency && typeof r.consistency.cr === 'number') crValue = r.consistency.cr;
      else if (typeof r.cr_mean === 'number') crValue = r.cr_mean;
      totalCR += crValue;
      const isExplicitlyBad = r.status === 'SUSPEITO' || r.status === 'CRÍTICO';
      const isMathematicallyBad = crValue > 0.2;
      const isMathematicallySuspect = crValue > 0.1;
      if (isExplicitlyBad || isMathematicallyBad) acc.critical++;
      else if (isMathematicallySuspect) acc.warning++;
      else acc.valid++;
      acc.total++;
      return acc;
    },
    { valid: 0, warning: 0, critical: 0, total: 0 }
  );
  acc.avgCR = acc.total > 0 ? totalCR / acc.total : 0;
  return acc;
}

const HEX_PAINEL = 'b'.repeat(64);
const shaDe = (id: string) => id.padEnd(64, '0').replace(/[^0-9a-f]/g, 'c');

function calculo(ids: string[] | undefined): any {
  const metadata: any = {
    judgmentsDigest: { algorithm: 'sha256', serialization: 'a12-julgamentos-v1', panel: HEX_PAINEL, unavailableReason: null },
  };
  if (ids !== undefined) {
    metadata.includedRespondents = ids.map((id) => ({
      respondentId: id,
      responseDocId: `doc-${id}`,
      identifierSource: 'respondentId',
      judgmentsSha256: shaDe(id),
      judgmentsUnavailableReason: null,
    }));
  }
  return { executionId: 'exec-fiacao-0001', metadata };
}

const ESTADOS = ['indisponivel', 'invalido', 'vinculado', 'divergente'] as const;

/** Um vínculo de cada estado, sobre a MESMA lista avaliada `lista`. ⚠ Em `divergente` a sobra é só do documento. */
function vinculoDoEstado(estado: (typeof ESTADOS)[number], lista: any[]) {
  const ids = lista.map((r) => r.respondentId);
  const doc =
    estado === 'indisponivel' ? calculo(undefined)
    : estado === 'invalido' ? calculo([])
    : estado === 'vinculado' ? calculo(ids)
    : calculo([...ids, 'z-so-no-documento']);
  const preparo = prepararVinculoDaTela({ calculo: doc, respondentesAtivos: lista, respostasAtivas: [] });
  expect(preparo.vinculo.estado).toBe(estado);
  expect(preparo.respondentesEnviados).toEqual(lista); // ⚠ o MESMO conjunto avaliado nos quatro estados
  return preparo.vinculo;
}

/** O payload como `page.tsx:1236-1315` o monta. `comEstatisticas: false` retira byStatus, summary e overallStats. */
function payloadReal(respondentes: any[], opcoes: { comVinculo: boolean; comEstatisticas?: boolean }) {
  const stats = individualStatsDaTela(respondentes);
  const respondentesComCR = respondentes.map((r: any) => ({
    ...r,
    cr: typeof r.cr === 'number' ? r.cr : (r.metrics?.avgCR ?? r.consistency?.cr ?? r.cr_mean ?? 0),
    isSimulated: r.isSimulated === true,
  }));
  const avaliacao = classificarAvaliacaoDaTela({ respondentesAvaliados: respondentes, respostasAtivas: [] });
  const avaliada = avaliacao.estado === 'disponivel';
  const comEstatisticas = opcoes.comEstatisticas !== false;
  const p: any = {
    avaliacaoDeQualidade: avaliacao,
    projectName: 'Projeto da medicao',
    bocrWeights: { Benefits: 0.37, Opportunities: 0.15, Costs: 0.2, Risks: 0.28 },
    bocrConsistency: { cr: 0.0106, lambda: 4.03 },
    responseCount: respondentes.length,
    finalScores: [
      { code: 'A1', name: 'Alternativa 1', score: 0.62 },
      { code: 'A2', name: 'Alternativa 2', score: 0.38 },
    ],
    qualityAnalysis: avaliada
      ? {
          respondents: respondentesComCR,
          ...(comEstatisticas
            ? {
                statistics: {
                  byStatus: { 'CONFIÁVEL': stats.valid, 'REVISAR': 0, 'SUSPEITO': stats.warning, 'CRÍTICO': stats.critical },
                  total: stats.total,
                  avgCR: stats.avgCR,
                },
                summary: { total: stats.total, ok: stats.valid, suspicious: stats.warning, critical: stats.critical },
              }
            : {}),
          overall: {},
        }
      : { respondents: respondentesComCR },
    overallStats: avaliada && comEstatisticas ? { total: stats.total, valid: stats.valid, warning: stats.warning, critical: stats.critical } : undefined,
    individualStats: avaliada && stats.total > 0 ? stats : undefined,
  };
  if (opcoes.comVinculo) p.vinculoDaExecucao = vinculoDoEstado('vinculado', respondentesComCR);
  return { payload: p, stats, avaliacao };
}

/**
 * ⚠ O CLASSIFICADOR INDEPENDENTE: a tabela da seção 2 do prompt, escrita aqui à mão, e NÃO a função do módulo
 * (que ainda não existe na base). Uma contagem é VÁLIDA quando é número inteiro finito e não negativo.
 */
const valida = (x: any): boolean => typeof x === 'number' && Number.isFinite(x) && Number.isInteger(x) && x >= 0;
function classificar(cob: any): { classe: 'a' | 'b' | 'c' | 'd'; condicao: number | null } {
  if (!cob || cob.restringiu !== true) return { classe: 'a', condicao: null };
  const A = cob.avaliadosAntesDaRestricao;
  const E = cob.enviados;
  const B = cob.enviadosDiferemDosAvaliados;
  const ambas = valida(A) && valida(E);
  if (B === false && ambas && E === A) return { classe: 'b', condicao: null };
  if (B === true && ambas && E < A) return { classe: 'c', condicao: null };
  if (ambas && E > A) return { classe: 'd', condicao: 1 };
  if (typeof B !== 'boolean') return { classe: 'd', condicao: 2 };
  if (!ambas) return { classe: 'd', condicao: 3 };
  if ((E !== A) !== B) return { classe: 'd', condicao: 4 };
  return { classe: 'd', condicao: 0 }; // inalcançável pela tabela: se sair, a tabela não é exaustiva
}

type Cap = { nome: string; grupo: string; payload: any; esperado?: string };

/** O vínculo do PRODUTOR REAL, entregue ao tratador com a lista enviada (ou com a `apresentada`, se difere). */
function cenarioProdutor(nome: string, avaliados: (string | null)[], doc: string[] | undefined, apresentada?: string[]): Cap {
  const prep = produtor(avaliados, doc);
  const lista = apresentada ? apresentada.map((a) => el(a)) : prep.respondentesEnviados;
  const { payload } = payloadReal(lista, { comVinculo: false });
  payload.vinculoDaExecucao = prep.vinculo;
  return { nome, grupo: 'produtor-real', payload, esperado: classificar(prep.vinculo.cobertura).classe };
}

  mexer(v.cobertura);

}

/** ⚠ O corpo da resposta traz um `timestamp` de execução (medido: é a ÚNICA linha que difere entre duas execuções do
 *   MESMO código, nos 56 corpos). Ele sai antes do hash; o resto do corpo se compara inteiro. */
const semVolateis = (x: any): any => {
  if (Array.isArray(x)) return x.map(semVolateis);
  if (x && typeof x === 'object') {
    const o: any = {};
    for (const k of Object.keys(x)) { if (k !== 'timestamp') o[k] = semVolateis(x[k]); }
    return o;
  }
  return x;
};

async function medir(c: Cap) {
  const r = await executar(c.payload);
  const linhas = linhasDeCobertura(r.contexto);
  gravar(`ctx/${c.nome}.txt`, r.contexto);
  gravar(`corpo/${c.nome}.json`, JSON.stringify(semVolateis(r.corpo), null, 1));
  const cobChega = c.payload.vinculoDaExecucao && typeof c.payload.vinculoDaExecucao === 'object' ? JSON.parse(JSON.stringify(c.payload.vinculoDaExecucao)).cobertura : undefined;
  return {
    nome: c.nome,
    grupo: c.grupo,
    status: r.status,
    classeDoVinculoQueChega: c.payload.vinculoDaExecucao === undefined || c.payload.vinculoDaExecucao === null ? 'sem-campo' : (() => { const k = classificar(cobChega); return k.classe + (k.condicao ? String(k.condicao) : ''); })(),
    esperado: c.esperado ?? null,
    linhasDeCobertura: linhas,
    sha256Contexto: sha256(r.contexto),
    bytesDoContexto: Buffer.byteLength(r.contexto, 'utf8'),
    sha256System: sha256(r.system),
    sha256Corpo: sha256(JSON.stringify(semVolateis(r.corpo))),
  };
}

describe('sonda: enumeração do produtor real', () => {
  test('toda a combinação de listas avaliadas e documentos pequenos', () => {
    const simbolos: (string | null)[] = ['r1', 'r2', 'r3', null];
    const listas: (string | null)[][] = [];
    const gerar = (n: number, atual: (string | null)[]) => {
      if (atual.length === n) { listas.push(atual); return; }
      for (const s of simbolos) gerar(n, [...atual, s]);
    };
    for (let n = 1; n <= 4; n++) gerar(n, []);
    expect(listas.length).toBe(340);

    const base = ['r1', 'r2', 'r3', 'r4'];
    const subconjuntos: string[][] = [];
    for (let m = 1; m < 16; m++) subconjuntos.push(base.filter((_, i) => (m >> i) & 1));
    expect(subconjuntos.length).toBe(15);

    const docs: Array<{ rotulo: string; calculo: any }> = [
      { rotulo: 'indisponivel', calculo: calculo(undefined) },
      { rotulo: 'invalido-vazio', calculo: calculo([]) },
      { rotulo: 'invalido-repetido', calculo: calculo(['r1', 'r1']) },
      { rotulo: 'invalido-nao-array', calculo: { ...calculo(['r1']), metadata: { ...calculo(['r1']).metadata, includedRespondents: 'x' } } },
      ...subconjuntos.map((s) => ({ rotulo: `ids:${s.join(',')}`, calculo: calculo(s) })),
    ];
    expect(docs.length).toBe(19);

    const origens: Array<{ origem: string; ativos: any[]; respostas: any[] }> = [
      ...listas.map((l) => ({ origem: 'analiseDeQualidade', ativos: l.map((x) => (x === null ? {} : { respondentId: x })), respostas: [] })),
      ...listas.map((l) => ({ origem: 'fallbackSobreRespostas', ativos: [], respostas: l.map((x) => (x === null ? {} : { id: x })) })),
      { origem: 'nenhuma', ativos: [], respostas: [] },
    ];
    expect(origens.length).toBe(681);

    const contagem: Record<string, number> = { a: 0, b: 0, c: 0, d: 0 };
    const porOrigem: Record<string, Record<string, number>> = {};
    const violacoes: Record<string, number> = { I1_enviadosMaiorQueAvaliados: 0, I2_booleanoDiscordaDasContagens: 0, I3_listaEnviadaDiferenteDaContagem: 0, I4_restringiuDiscordaDoEstado: 0, I5_retiradoSemSobraNaAvaliacao: 0, I6_sobraSemRetirada: 0, T_tabelaNaoExaustiva: 0 };
    const estadosEmB: Record<string, number> = { vinculado: 0, divergente: 0 };
    const retiradosEmC: Record<string, number> = {};
    const exemplos: Record<string, any[]> = { a: [], b: [], c: [], d: [] };
    const linhaAntes = { aberturaAntigaEmB: 0, aberturaAntigaEmC: 0, aberturaAEmA: 0, outras: 0 };
    const linhaDepois = { esperadaB: 0, esperadaC: 0, esperadaA: 0, esperadaD: 0, divergiu: 0, exemplosDeDivergencia: [] as any[] };

    let n = 0;
    for (const o of origens) {
      for (const d of docs) {
        const prep = prepararVinculoDaTela({ calculo: d.calculo, respondentesAtivos: o.ativos, respostasAtivas: o.respostas });
        const v: any = prep.vinculo;
        const cob = v.cobertura;
        const k = classificar(cob);
        contagem[k.classe]++;
        porOrigem[o.origem] = porOrigem[o.origem] ?? { a: 0, b: 0, c: 0, d: 0 };
        porOrigem[o.origem][k.classe]++;
        n++;
        if (k.classe === 'd' && k.condicao === 0) violacoes.T_tabelaNaoExaustiva++;
        // invariantes lidos do código, e MEDIDOS aqui
        if (cob.restringiu === true) {
          if (cob.enviados > cob.avaliadosAntesDaRestricao) violacoes.I1_enviadosMaiorQueAvaliados++;
          if (cob.enviadosDiferemDosAvaliados !== (cob.enviados !== cob.avaliadosAntesDaRestricao)) violacoes.I2_booleanoDiscordaDasContagens++;
          const enviadaDeFato = o.origem === 'analiseDeQualidade' ? prep.respondentesEnviados.length : prep.respostasEnviadas.length;
          if (enviadaDeFato !== cob.enviados) violacoes.I3_listaEnviadaDiferenteDaContagem++;
          const sobra = (v.divergencia.sobraNaAvaliacao ?? []).length;
          if (cob.avaliadosAntesDaRestricao - cob.enviados !== sobra) violacoes.I5_retiradoSemSobraNaAvaliacao++;
          if (k.classe === 'b' && sobra !== 0) violacoes.I6_sobraSemRetirada++;
        }
        if ((v.estado === 'vinculado' || v.estado === 'divergente') !== (cob.restringiu === true)) violacoes.I4_restringiuDiscordaDoEstado++;
        if (k.classe === 'b') estadosEmB[v.estado]++;
        if (k.classe === 'c') { const q = String(cob.avaliadosAntesDaRestricao - cob.enviados); retiradosEmC[q] = (retiradosEmC[q] ?? 0) + 1; }
        if (exemplos[k.classe].length < 3) exemplos[k.classe].push({ origem: o.origem, doc: d.rotulo, estado: v.estado, cobertura: cob });

        // a linha, como a descrição a escreve
        const linha = descreverVinculoParaContexto(v).split('\n').find((l) => l.startsWith(PREFIXO)) ?? '';
        if (FASE === 'antes') {
          if (k.classe === 'a' && linha === ABERTURA_A) linhaAntes.aberturaAEmA++;
          else if (k.classe === 'b' && linha.startsWith(ABERTURA_ANTIGA)) linhaAntes.aberturaAntigaEmB++;
          else if (k.classe === 'c' && linha.startsWith(ABERTURA_ANTIGA)) linhaAntes.aberturaAntigaEmC++;
          else linhaAntes.outras++;
        } else {
          const ok =
            (k.classe === 'a' && linha === ABERTURA_A) ||
            (k.classe === 'b' && linha.startsWith(ABERTURA_B)) ||
            (k.classe === 'c' && linha.startsWith(ABERTURA_C)) ||
            (k.classe === 'd' && linha.startsWith(ABERTURA_D));
          if (ok) { if (k.classe === 'a') linhaDepois.esperadaA++; else if (k.classe === 'b') linhaDepois.esperadaB++; else if (k.classe === 'c') linhaDepois.esperadaC++; else linhaDepois.esperadaD++; }
          else { linhaDepois.divergiu++; if (linhaDepois.exemplosDeDivergencia.length < 3) linhaDepois.exemplosDeDivergencia.push({ classe: k.classe, linha }); }
        }
      }
    }
    expect(n).toBe(681 * 19);
    gravar(`enumeracao-${FASE}.json`, JSON.stringify({ fase: FASE, combinacoes: n, origens: origens.length, documentos: docs.length, contagem, porOrigem, estadosEmB, retiradosEmC, violacoes, linhaAntes, linhaDepois, exemplos }, null, 2));
  });
});
