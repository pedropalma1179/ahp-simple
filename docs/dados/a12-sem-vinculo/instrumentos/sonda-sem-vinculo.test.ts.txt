async function executar(payload: any, modo: 'captura' | 'texto-fixado', texto = ''): Promise<Execucao> {
  estado.modo = modo;
  estado.texto = texto;
  estado.contexto = '';
  estado.system = '';
  estado.args = null;
  logs = [];
  const antes = estado.chamadas;
  const redeAntes = chamadasDeRede.length;
  const jsonEntregue = JSON.stringify(payload); // ⚠ O MESMO transporte da tela (`page.tsx:1353`)
  const request = new Request('http://localhost/api/ai-reviewer', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: jsonEntregue,
  });
  const textoLido = await request.clone().text(); // o que o tratador receberia como TEXTO
  const res = await POST(request as any); // o tratador faz `await request.json()`
  const textoDoCorpo = await res.text();
  return {
    jsonEntregue,
    textoLidoDoRequestIgual: textoLido === jsonEntregue,
    status: res.status ?? 200,
    textoDoCorpo,
    contexto: estado.contexto,
    system: estado.system,
    args: estado.args,
    chamadasAoCliente: estado.chamadas - antes,
    chamadasDeRede: chamadasDeRede.length - redeAntes,
  };
}

function descricaoDoValor(p: any): string {
  if (!Object.prototype.hasOwnProperty.call(p, 'vinculoDaExecucao')) return 'chave ausente no objeto preparado';
  const v = p.vinculoDaExecucao;
  if (v === undefined) return 'chave PRESENTE no objeto preparado, com valor undefined';
  if (v === null) return 'chave presente, com valor null';
  return `chave presente, com objeto (estado "${v.estado}")`;
}

/** Os caminhos de chave de um valor JSON, recursivamente (arrays por `[]`), e os que contêm o radical do vínculo. */
function caminhosDeChave(v: any, prefixo = '', acc: string[] = []): string[] {
  if (Array.isArray(v)) {
    if (v.length === 0) acc.push(`${prefixo}[]`);
    for (const x of v) caminhosDeChave(x, `${prefixo}[]`, acc);
  } else if (v !== null && typeof v === 'object') {
    const ks = Object.keys(v);
    if (ks.length === 0) acc.push(`${prefixo}{}`);
    for (const k of ks) {
      acc.push(prefixo ? `${prefixo}.${k}` : k);
      caminhosDeChave(v[k], prefixo ? `${prefixo}.${k}` : k, acc);
    }
  }
  return acc;
}

type No = any;
const filhos = (no: No): No[] => {
  const c = no?.props?.children;
  return c === undefined || c === null ? [] : Array.isArray(c) ? (c as any[]).flat(Infinity) : [c];
};
function achar(no: No, pred: (n: No) => boolean, acc: No[] = []): No[] {
  if (no && typeof no === 'object' && !Array.isArray(no)) {
    if (pred(no)) acc.push(no);
    for (const f of filhos(no)) achar(f, pred, acc);
  } else if (Array.isArray(no)) {
    for (const f of (no as any[]).flat(Infinity)) achar(f, pred, acc);
  }
  return acc;
}
function textoDe(no: No): string {
  if (no === null || no === undefined || typeof no === 'boolean') return '';
  if (typeof no === 'string' || typeof no === 'number') return String(no);
  if (Array.isArray(no)) return (no as any[]).map(textoDe).join('');
  return filhos(no).map(textoDe).join('');
}

            respostaDoTratadorNaCaptura: { status: ex.status, textoDoCorpo: ex.textoDoCorpo, observacao: 'erro sentinela do cliente simulado: a captura interrompe antes da geracao' },
            contexto: metaDoContexto(ex, `fase-a/contextos/ctx-${id}.txt`),
            chamadaAoCliente: { n: ex.chamadasAoCliente, argumentos: ex.args, systemSha256: sha256(ex.system), systemBytes: Buffer.byteLength(ex.system, 'utf8') },
            chamadasDeRede: ex.chamadasDeRede,
          });
        }
      }
    }
    passadas.push(m);
  }
  const determinismoA = Object.keys(passadas[0]).map((id) => ({
    id,
    contextoIgualNasDuasPassadas: passadas[0][id].contextoSha === passadas[1][id].contextoSha,
    jsonEntregueIgualNasDuasPassadas: passadas[0][id].jsonSha === passadas[1][id].jsonSha,
  }));

  // ------------------------------------------------------------------------------------------ FASE B: quatro execuções com texto fixado
  const registrosB: any[] = [];
  const corposDaPassada1: Record<string, string> = {};
  const passadasB: Array<Record<string, { corpoSemTimestampSha: string; corpoSha: string; contextoSha: string }>> = [];
  for (const passada of [1, 2]) {
    const m: Record<string, { corpoSemTimestampSha: string; corpoSha: string; contextoSha: string }> = {};
    for (const cl of CLASSES) {
      for (const v of ['controle', 'chave-ausente'] as Variante[]) {
        const id = `${cl}__${v}`;
        const payload = montarPayload('sem-exclusao', v, cl);
        const ex = await executar(payload, 'texto-fixado', TEXTO_FIXADO);
        expect(ex.status).toBe(200);
        expect(ex.chamadasAoCliente).toBe(1);
        expect(ex.chamadasDeRede).toBe(0);
        expect(ex.textoLidoDoRequestIgual).toBe(true);
        const corpo = JSON.parse(ex.textoDoCorpo);
        const semTimestamp = JSON.parse(ex.textoDoCorpo);
        delete semTimestamp.metadata.timestamp; // ⚠ ÚNICA exclusão autorizada, e só para a PROJEÇÃO de comparação
        m[id] = { corpoSemTimestampSha: sha256(JSON.stringify(semTimestamp)), corpoSha: sha256(ex.textoDoCorpo), contextoSha: sha256(ex.contexto) };
        if (passada === 1) {
          corposDaPassada1[id] = ex.textoDoCorpo;
          escrever(`fase-b/corpos/corpo-${id}.json`, ex.textoDoCorpo);
          escrever(`fase-b/contextos/ctx-${id}.txt`, ex.contexto);
          escrever(`fase-b/jsons-entregues/${id}.json`, ex.jsonEntregue);
          const caminhos = caminhosDeChave(corpo);
          const mencoesNasChaves = caminhos.filter((c) => RADICAL.test(c));
          const mencoesNosValores: string[] = [];
          const varrer = (x: any, p: string) => {
            if (typeof x === 'string') { if (RADICAL.test(x)) mencoesNosValores.push(p); }
            else if (Array.isArray(x)) x.forEach((y, i) => varrer(y, `${p}[${i}]`));
            else if (x && typeof x === 'object') for (const k of Object.keys(x)) varrer(x[k], p ? `${p}.${k}` : k);
          };
          varrer(corpo, '');
          registrosB.push({
            id,
            classificacao: cl,
            variante: v,
            condicaoDeExclusao: 'sem-exclusao',
            jsonEntregue: { arquivo: `fase-b/jsons-entregues/${id}.json`, sha256: sha256(ex.jsonEntregue), bytes: Buffer.byteLength(ex.jsonEntregue, 'utf8'), contemAChaveVinculoNoTexto: ex.jsonEntregue.includes('"vinculoDaExecucao"') },
            corpo: { arquivo: `fase-b/corpos/corpo-${id}.json`, sha256: sha256(ex.textoDoCorpo), bytes: Buffer.byteLength(ex.textoDoCorpo, 'utf8'), status: ex.status, timestamp: corpo.metadata?.timestamp ?? null },
            chavesDeNivelPrincipal: Object.keys(corpo),
            chavesDeMetadata: Object.keys(corpo.metadata ?? {}),
            caminhosDeChave: caminhos,
            mencoesAoRadicalVinculoNosCaminhosDeChave: mencoesNasChaves,
            mencoesAoRadicalVinculoNosValoresDeTexto: mencoesNosValores,
            nota: corpo.nota,
            veredicto: corpo.veredicto,
            estadoDaExtracao: corpo.metadata?.estadoDaExtracao ?? null,
            mensagemDaExtracao: corpo.mensagemDaExtracao ?? null,
            notaSuspensa: corpo.notaSuspensa === null ? null : { suspensa: corpo.notaSuspensa?.suspensa, causa: corpo.notaSuspensa?.causa, rotulo: corpo.notaSuspensa?.rotulo, motivo: corpo.notaSuspensa?.motivo },
            validacaoA27: { version: corpo.validation?.version, estado: corpo.validation?.estado, isValid: corpo.validation?.isValid, nIssues: (corpo.validation?.issues ?? []).length, nWarnings: (corpo.validation?.warnings ?? []).length, nInconclusivos: (corpo.validation?.inconclusivos ?? []).length },
            reviewIgualAoTextoFixado: corpo.review === TEXTO_FIXADO,
            contexto: metaDoContexto(ex, `fase-b/contextos/ctx-${id}.txt`),
            chamadaAoCliente: { n: ex.chamadasAoCliente, argumentos: ex.args },
            chamadasDeRede: ex.chamadasDeRede,
          });
        }
      }
    }
    passadasB.push(m);
  }
  const determinismoB = Object.keys(passadasB[0]).map((id) => ({
    id,
    corpoIgualNasDuasPassadas: passadasB[0][id].corpoSha === passadasB[1][id].corpoSha,
    corpoSemTimestampIgualNasDuasPassadas: passadasB[0][id].corpoSemTimestampSha === passadasB[1][id].corpoSemTimestampSha,
    contextoIgualNasDuasPassadas: passadasB[0][id].contextoSha === passadasB[1][id].contextoSha,
  }));
  // O contexto da Fase B `elegivel` (sem exclusao) tem de ser o da Fase A (sem exclusao), com o mesmo controle/ausente.
  const cruzamentoAB = (['controle', 'chave-ausente'] as Variante[]).map((v) => ({
    variante: v,
    contextoDaFaseBElegivelIgualAoDaFaseAEmSemExclusao: registrosB.find((r) => r.id === `elegivel__${v}`).contexto.sha256 === registrosA.find((r) => r.id === `sem-exclusao__${v}`).contexto.sha256,
    sha256DaFaseB: registrosB.find((r) => r.id === `elegivel__${v}`).contexto.sha256,
    sha256DaFaseA: registrosA.find((r) => r.id === `sem-exclusao__${v}`).contexto.sha256,
  }));

  // ------------------------------------------------------------------------------------------ FASE C: renderização do componente com os quatro corpos
  const registrosC: any[] = [];
  for (const r of registrosB) {
    const corpo = JSON.parse(corposDaPassada1[r.id]);
    // ⚠ O `aiReview` é o que a tela monta em `page.tsx:1363-1371`: sete campos, e nem mais nem menos.
    const aiReview = {
      nota: corpo.nota,
      veredicto: corpo.veredicto,
      notaSuspensa: corpo.notaSuspensa ?? null,
      mensagemDaExtracao: corpo.mensagemDaExtracao ?? null,
      review: corpo.review,
      validation: corpo.validation,
      metadata: corpo.metadata,
    };
    const apresentacao = decideReviewPresentation(aiReview.validation);
    const s = superficies(aiReview);
    escrever(`fase-c/html/${r.id}.html`, s.html + '\n');
    registrosC.push({
      id: r.id,
      classificacao: r.classificacao,
      variante: r.variante,
      aiReview: { chaves: Object.keys(aiReview), sha256DoJson: sha256(JSON.stringify(aiReview)) },
      estadoDeA27: { estado: apresentacao.estado, quarantined: apresentacao.quarantined, titulo: apresentacao.titulo, nMotivos: apresentacao.motivos.length, nWarnings: apresentacao.warnings.length },
      portaoDoDestaque: {
        condicao1_classificacaoNaoSuspensa: !aiReview.notaSuspensa?.suspensa,
        condicao2_apresentacaoAprovada: apresentacao.estado === 'aprovado',
        condicao3_notaOuVeredictoNaoNulos: !!(aiReview.nota || aiReview.veredicto),
      },
      renderizacao: {
        html: { arquivo: `fase-c/html/${r.id}.html`, sha256: sha256(s.html), caracteres: s.html.length },
        destaque: s.destaque,
        avisoDeQuarentena: s.avisoDeQuarentena,
        avisosDaVerificacao: s.avisosDaVerificacao,
        cartaoDeSuspensao: s.cartaoDeSuspensao,
        mensagemDaExtracao: s.mensagemDaExtracao,
        titulosDeBlocosDeStatus: s.titulosDeBlocosDeStatus,
        corpoDoTexto: s.corpoDoTexto,
        mencoesAoRadicalVinculoNoHtml: (s.html.match(/v[ií]nculo/gi) || []).length,
      },
    });
  }
  const comparacaoC = CLASSES.map((cl) => {
    const c = registrosC.find((x) => x.id === `${cl}__controle`);
    const a = registrosC.find((x) => x.id === `${cl}__chave-ausente`);
    return {
      classificacao: cl,
      htmlIgualEntreControleEAusente: c.renderizacao.html.sha256 === a.renderizacao.html.sha256,
      sha256DoHtmlControle: c.renderizacao.html.sha256,
      sha256DoHtmlAusente: a.renderizacao.html.sha256,
    };
  });

  // ------------------------------------------------------------------------------------------ A IDENTIDADE
  const arquivosDeProducao = [
    'app/api/ai-reviewer/route.ts',
    'lib/ai-reviewer/vinculo-execucao.ts',
    'lib/ai-reviewer/avaliacao-qualidade.ts',
    'lib/ai-reviewer/review-validation-contract.ts',
    'lib/ai-reviewer/review-request.ts',
    'components/ParecerAISection.tsx',
    'app/decisor/resultados/[projectId]/page.tsx',
  ];
  const identidade = {
    natureza: 'identidade da execucao da sonda do caminho sem vinculoDaExecucao; os valores sao MEDIDOS nesta execucao, salvo onde dito',
    instanteDaMedicaoUTC: instanteDaMedicao,
    codigo: {
      commitDeBase: execGit('git rev-parse HEAD'),
      statusPorcelainAntesDaExecucao: execGit('git status --porcelain'),
      sha256DosArquivosDeProducao: Object.fromEntries(arquivosDeProducao.map((f) => [f, sha256Arq(f)])),
      instrumento: { arquivoExecutado: path.basename(__filename), sha256: sha256(fs.readFileSync(__filename, 'utf8')) },
    },
    ambiente: {
      node: process.version,
      plataforma: `${process.platform} ${process.arch}`,
      npm: execGit('npm -v'),
      // ⚠ Duas fontes que NAO coincidem, e as duas se conservam: a versao declarada nos `package.json` instalados, e a que o proprio CLI imprime.
      versaoDosPacotesInstalados: Object.fromEntries(
        ['jest', 'jest-cli', '@jest/core', 'ts-jest', 'react', 'react-dom', 'next', '@anthropic-ai/sdk'].map((p) => [p, JSON.parse(fs.readFileSync(path.join(RAIZ, 'node_modules', p, 'package.json'), 'utf8')).version])
      ),
      jestVersionImpressaPeloCli: execGit('node node_modules/.bin/jest --version'),
    },
    modelo: {
      real: 'NAO APLICAVEL: nenhum modelo foi chamado; o cliente de `@anthropic-ai/sdk` foi SIMULADO (jest.mock) e devolveu um erro sentinela (Fase A) ou o texto fixado (Fase B)',
      parametrosDaChamadaCapturados: registrosA[0].chamadaAoCliente.argumentos,
      parametrosNaoEnviados: 'temperature, top_p e top_k: conferido na lista de chaves da chamada capturada (nao constam); seed: nao existe na chamada',
    },
    dados: { natureza: 'SINTETICOS: quatro respondentes (CR 0.05), documento de calculo sintetico, hashes derivados de etiquetas; a tela nao foi executada', idsDosRespondentes: IDS, textoFixadoDaFaseB: { sha256: sha256(TEXTO_FIXADO), caracteres: TEXTO_FIXADO.length } },
    variaveisDeAmbiente: { ANTHROPIC_API_KEY: 'chave falsa do processo de teste', USE_RAG_SEMANTIC: 'removida (RAG semantico desligado)' },
    rede: { chamadasDeFetchObservadas: chamadasDeRede.length, observacao: 'fetch global substituido por funcao que lanca e conta; nenhuma chamada observada' },
    ausenciasDeMetadadoDeclaradas: [
      'temperatura, top_p, top_k e seed: nao se aplicam (nao enviados)',
      'versao do modelo real: nao se aplica (cliente simulado)',
      'nenhum registro de latencia, custo ou tokens: nao se aplica',
    ],
  };

  escrever('fase-a/capturas.json', JSON.stringify({ natureza: 'FASE A, captura interrompida: o contexto que SERIA enviado ao modelo; oito capturas (2 condicoes de exclusao x 4 variantes do campo)', registros: registrosA, determinismoEntreDuasPassadas: determinismoA }, null, 2) + '\n');
  escrever('fase-b/execucoes.json', JSON.stringify({ natureza: 'FASE B, texto fixado e o tratador concluindo: o corpo da resposta; quatro execucoes (2 classificacoes x controle/chave ausente), condicao de exclusao fixa: sem exclusao', registros: registrosB, determinismoEntreDuasPassadas: determinismoB, cruzamentoComAFaseA: cruzamentoAB }, null, 2) + '\n');
  escrever('fase-c/renderizacoes.json', JSON.stringify({ natureza: 'FASE C, renderizacao do componente real ParecerAISection com os quatro corpos da Fase B; NAO e a pagina no navegador', registros: registrosC, comparacaoControleEAusente: comparacaoC }, null, 2) + '\n');
  escrever('identidade.json', JSON.stringify(identidade, null, 2) + '\n');
  expect(chamadasDeRede.length).toBe(0);
  expect(determinismoA.every((d) => d.contextoIgualNasDuasPassadas && d.jsonEntregueIgualNasDuasPassadas)).toBe(true);
});
