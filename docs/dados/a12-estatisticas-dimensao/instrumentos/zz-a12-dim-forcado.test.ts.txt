async function executar(payload: any) {

const item = (id: string, cr: number) => ({ respondentId: id, id, name: `R ${id}`, cr, metrics: { avgCR: cr }, status: cr <= 0.1 ? 'CONFIÁVEL' : cr <= 0.15 ? 'REVISAR' : cr <= 0.2 ? 'SUSPEITO' : 'CRÍTICO' });

const linhaTaxa = (c: string) => c.split('\n').find((l) => l.startsWith('**Taxa de Validade Geral:**'));
const bloco = (c: string) => { const L = c.split('\n'); const i = L.findIndex((l) => l.startsWith('## Estatísticas por Dimensão BOCR')); const f = L.findIndex((l, k) => k > i && l.startsWith('## Pesos Finais')); return L.slice(i, f).join('\n'); };

test('ramo dos totais que SOMA as quatro dimensoes, com a elegibilidade forcada', async () => {
  const saida: any[] = [];
  for (const n of [3, 5, 6, 12]) {
    const c = await executar(base(n, { valid: n, warning: 0, critical: 0, total: n, avgCR: 0.05 }));
    saida.push({ caso: `agregado ${n}, todos validos`, taxa: linhaTaxa(c), primeiraDimensao: bloco(c).split('\n').slice(2, 6) });
  }
  // agregado 5 com a distribuicao do exemplo (2 validos, 1 warning, 2 criticos)
  const c5 = await executar(base(5, { valid: 2, warning: 1, critical: 2, total: 5, avgCR: 0.136 }));
  saida.push({ caso: 'agregado 5: 2 validos, 1 warning, 2 criticos', taxa: linhaTaxa(c5), primeiraDimensao: bloco(c5).split('\n').slice(2, 6) });
  // por dimensao, em PESSOAS (o que uma saida B enviaria): 5 pessoas, validas 5, 4, 3, 2
  const cB = await executar(base(5, {
    Benefits: { total: 5, valid: 5, warning: 0, critical: 0 },
    Opportunities: { total: 5, valid: 4, warning: 1, critical: 0 },
    Costs: { total: 5, valid: 3, warning: 1, critical: 1 },
    Risks: { total: 5, valid: 2, warning: 2, critical: 1 },
  }));
  saida.push({ caso: 'por dimensao em pessoas (5 pessoas; validas 5,4,3,2)', taxa: linhaTaxa(cB), somaDasQuatro: { total: 20, validas: 14 } });
  // sem nenhum numero por dimensao (o que uma saida A enviaria): individualStats ausente
  const cA = await executar(base(5, undefined));
  saida.push({ caso: 'sem individualStats (o que uma saida A enviaria)', taxa: linhaTaxa(cA), bloco: bloco(cA).split('\n').slice(0, 6) });
  fs.mkdirSync(process.env.SAIDA_DIM as string, { recursive: true });
  fs.writeFileSync(`${process.env.SAIDA_DIM}/ramo4-forcado.json`, JSON.stringify(saida, null, 1));
});
