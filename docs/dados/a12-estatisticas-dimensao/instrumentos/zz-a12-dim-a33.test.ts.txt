const captura: { contexto: string } = { contexto: '' };
jest.mock('@anthropic-ai/sdk', () => ({

const { POST } = require('@/app/api/ai-reviewer/route');
let silencios: jest.SpyInstance[] = [];

const RAIZ = path.resolve(__dirname, '..', '..');

test('o bloco por dimensao e a Taxa nas requisicoes de referencia do a33 (sem avaliacaoDeQualidade e sem individualStats)', async () => {
  const saida: any[] = [];
  for (const nome of ['r1', 'r2']) {
    const d = JSON.parse(fs.readFileSync(path.join(RAIZ, `docs/dados/a33-etapa4-v2/requisicao-referencia-${nome}.json`), 'utf8'));
    captura.contexto = '';
    const res = await POST({ json: async () => JSON.parse(JSON.stringify(d.requisicao)) } as any);
    await res.json();
    const L = captura.contexto.split('\n');
    const i = L.findIndex((l) => l.startsWith('## Estatísticas por Dimensão BOCR'));
    const f = L.findIndex((l, k) => k > i && l.startsWith('## Pesos Finais da Hierarquia'));
    saida.push({
      nome,
      bytesDoContexto: Buffer.byteLength(captura.contexto, 'utf8'),
      taxa: L.filter((l) => l.startsWith('**Taxa de Validade Geral:**')),
      bloco: L.slice(i, f).join('\n'),
      totalDaLista: L.filter((l) => l.startsWith('**TOTAL')),
      respondentes: d.requisicao.qualityAnalysis.respondents.length,
      statisticsTotal: d.requisicao.qualityAnalysis.statistics?.total,
      byStatus: d.requisicao.qualityAnalysis.statistics?.byStatus,
    });
  }
  fs.mkdirSync(process.env.SAIDA_DIM as string, { recursive: true });
  fs.writeFileSync(`${process.env.SAIDA_DIM}/a33-referencia-bloco.json`, JSON.stringify(saida, null, 1));
});
