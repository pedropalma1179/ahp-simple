const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');
const [, , dir, saida] = process.argv;
const sha256 = (t) => crypto.createHash('sha256').update(t, 'utf8').digest('hex');
const ocorrencias = (t, a) => t.split(a).length - 1;

const TITULO = '**Referência Automatizada (apenas contexto — NÃO use como sua decisão):**\n';
const ULTIMA = '- IMPORTANTE: Sua DECISÃO EDITORIAL na seção 🎯 deve ser baseada na SUA análise dos dados, NÃO nesta referência automática.\n';
const PREFIXO_ANTIGO = 'SUSPENSA: Nota não calculada: ';
const PREFIXO_NOVO = 'SUSPENSA: ';

function T(ctx) {
  if (ocorrencias(ctx, TITULO) !== 1 || ocorrencias(ctx, ULTIMA) !== 1) throw new Error('bloco: ' + ocorrencias(ctx, TITULO) + '/' + ocorrencias(ctx, ULTIMA));
  const i = ctx.indexOf(TITULO);
  const fim = ctx.indexOf(ULTIMA, i) + ULTIMA.length;
  const bloco = ctx.slice(i, fim);
  const semBloco = ctx.slice(0, i) + ctx.slice(fim);
  const n = ocorrencias(semBloco, PREFIXO_ANTIGO);
  if (n > 1) throw new Error('prefixo ' + n);
  return { bloco, previsto: n === 1 ? semBloco.replace(PREFIXO_ANTIGO, PREFIXO_NOVO) : semBloco, suspensa: n === 1 };
}

    sha256Antigo: sha256(e.contexto), bytesAntigo: Buffer.byteLength(e.contexto, 'utf8'),
    blocoAntigo: t.bloco,

    sha256Previsto: sha256(t.previsto), bytesPrevisto: Buffer.byteLength(t.previsto, 'utf8'),
    coerente: t.suspensa === !e.elegivel,
  };
});

function blocoDeQualidade(texto) {
  const cab = ['**Distribuição de Qualidade dos Respondentes (CR Individual):**', '**Distribuição de Qualidade dos Respondentes:**', '**Qualidade dos Dados:**'].find((c) => texto.includes(c));
  if (!cab) return [];
  const linhas = texto.slice(texto.indexOf(cab)).split('\n');
  const bloco = [cab];
  for (const l of linhas.slice(1, 11)) { if (l.trim() === '') break; bloco.push(l); }
  return bloco;
}

  A12_MESSAGES_SEM_CHUNKS: '5b2cb948064279be81ae4488dd87dfc8b3ccc24e1c2563f2f4e684bde82243da',
