const PADROES: Array<{ pattern: RegExp; nota: string; veredicto: string }> = [
  { pattern: /\bACEITO\s+COM\s+REVIS(?:Õ|O)ES\s+MENORES\b/i, nota: 'A', veredicto: 'ACEITO COM REVISÕES MENORES' },
  { pattern: /\bREVIS(?:Õ|O)ES\s+MENORES\b/i, nota: 'B', veredicto: 'REVISÕES MENORES' },
  { pattern: /\bREVIS(?:Õ|O)ES\s+MAIORES\b/i, nota: 'C', veredicto: 'REVISÕES MAIORES' },
  { pattern: /\bRECONSIDERAR\b/i, nota: 'D', veredicto: 'RECONSIDERAR APÓS REVISÃO SUBSTANCIAL' },
  { pattern: /\bREJEITAR\b/i, nota: 'F', veredicto: 'REJEITAR' },
  { pattern: /\bACEITO\b/i, nota: 'A', veredicto: 'ACEITO' },
];
const REGEX_EDITORIAL = /DECIS(?:Ã|A)O\s+EDITORIAL[^\n]*\n([\s\S]{0,500})/i;

function replica(reviewText: string) {
  if (!reviewText || reviewText.length < 50) return { caminho: 'curto', indice: null as number | null, nota: null as string | null, veredicto: null as string | null };
  const m = reviewText.match(REGEX_EDITORIAL);
  if (m) {
    for (let i = 0; i < PADROES.length; i++) {
      if (PADROES[i].pattern.test(m[1])) return { caminho: 'editorial', indice: i, nota: PADROES[i].nota, veredicto: PADROES[i].veredicto };
    }
  }
  for (let i = 0; i < PADROES.length; i++) {
    if (PADROES[i].pattern.test(reviewText)) return { caminho: 'texto-completo', indice: i, nota: PADROES[i].nota, veredicto: PADROES[i].veredicto };
  }
  return { caminho: 'nenhum', indice: null as number | null, nota: null as string | null, veredicto: null as string | null };
}

function anatomia(texto: string) {
  const idxTitulo = texto.search(/DECIS(?:Ã|A)O\s+EDITORIAL/i);
  const m = texto.match(REGEX_EDITORIAL);
  const janela = m ? m[1] : null;
  const inicioDaJanela = m && m.index !== undefined ? m.index + m[0].length - m[1].length : null;
  return {
    extensaoEmCaracteres: texto.length,
    extensaoEmPontosDeCodigo: Array.from(texto).length,
    bytesUtf8: Buffer.byteLength(texto, 'utf8'),
    guardaDe50Satisfeita: texto.length >= 50,
    posicaoDoTituloDaSecaoEditorial: idxTitulo,
    secaoEditorialCapturadaPelaRegex: m !== null,
    inicioDaJanelaDe500: inicioDaJanela,
    janelaCapturada: janela,
    janelaExtensao: janela === null ? null : janela.length,
    padroesQueCasamNaJanela: janela === null ? [] : PADROES.map((p, i) => (p.pattern.test(janela) ? i : -1)).filter((i) => i >= 0),
    padroesQueCasamNoTextoInteiro: PADROES.map((p, i) => (p.pattern.test(texto) ? i : -1)).filter((i) => i >= 0),
  };
}

type Execucao = { corpo: any; status: number; chamadas: number; contexto: string; system: string; logs: string[] };

async function executar(payload: any, modo: 'captura' | 'texto-fixado', texto: string, coerenciaNula = false): Promise<Execucao> {
  estado.modo = modo;
  estado.texto = texto;
  estado.contexto = '';
  estado.system = '';
  estado.coerenciaNula = coerenciaNula;
  logs = [];
  const antes = estado.chamadas;
  const res = await POST({ json: async () => JSON.parse(JSON.stringify(payload)) } as any);
  const corpo = await res.json();
  estado.coerenciaNula = false;
  return { corpo, status: res.status ?? 200, chamadas: estado.chamadas - antes, contexto: estado.contexto, system: estado.system, logs: [...logs] };
}

type No = any;
const filhos = (no: No): No[] => {
  const c = no?.props?.children;
  return c === undefined || c === null ? [] : Array.isArray(c) ? (c as any[]).flat(Infinity) : [c];
};
function achar(no: No, pred: (n: No) => boolean, acc: No[] = []): No[] {
  if (no && typeof no === 'object' && !Array.isArray(no)) {
    if (pred(no)) acc.push(no);
    for (const f of filhos(no)) achar(f, pred, acc);
  } else if (Array.isArray(no)) {
    for (const f of (no as any[]).flat(Infinity)) achar(f, pred, acc);
  }
  return acc;
}
function textoDe(no: No): string {
  if (no === null || no === undefined || typeof no === 'boolean') return '';
  if (typeof no === 'string' || typeof no === 'number') return String(no);
  if (Array.isArray(no)) return (no as any[]).map(textoDe).join('');
  return filhos(no).map(textoDe).join('');
}

async function acionarCopia(arvore: No, setCopied: jest.Mock) {
  const botoes = achar(arvore, (n) => n.type === 'button' && n.props?.title === 'Copiar parecer em markdown');
  if (botoes.length === 0) return { botaoExibido: false, manipuladorAcionado: false, argumento: null as string | null, chamadasDeWriteText: 0, setCopiedChamadoCom: [] as unknown[] };
  const chamadas: string[] = [];
  const antes = Object.getOwnPropertyDescriptor(globalThis, 'navigator');
  Object.defineProperty(globalThis, 'navigator', {
    value: { clipboard: { writeText: async (t: string) => { chamadas.push(t); } } },
    configurable: true,
    writable: true,
  });
  const espiaoTimeout = jest.spyOn(globalThis, 'setTimeout').mockImplementation(((fn: any) => 0 as any) as any);
  try {
    await botoes[0].props.onClick();
  } finally {
    espiaoTimeout.mockRestore();
    if (antes) Object.defineProperty(globalThis, 'navigator', antes); else delete (globalThis as any).navigator;
  }
  return {
    botaoExibido: true,
    manipuladorAcionado: true,
    argumento: chamadas.length > 0 ? chamadas[0] : null,
    chamadasDeWriteText: chamadas.length,
    setCopiedChamadoCom: setCopied.mock.calls.map((c) => c[0]),
  };
}

async function superficies(aiReview: any) {
  const { arvore, setCopied } = arvoreDoComponente(aiReview);
  const html: string = renderToStaticMarkup(React.createElement(ParecerAISection as any, { aiReview, loading: false, onExecute: () => undefined }));

  // 1. DESTAQUE
  const caixaDestaque = achar(arvore, (n) => classe(n).includes('flex gap-3 mb-6 flex-wrap'));
  const paragrafos = caixaDestaque.length ? achar(caixaDestaque[0], (n) => n.type === 'p').map((n) => textoDe(n)) : [];
  const idxNota = paragrafos.indexOf('Nota Científica');
  const idxVered = paragrafos.indexOf('Veredicto');
  const destaque = {
    exibido: caixaDestaque.length > 0,
    nota: idxNota >= 0 ? paragrafos[idxNota + 1] ?? null : null,
    veredicto: idxVered >= 0 ? paragrafos[idxVered + 1] ?? null : null,
  };

  // 2. AVISO (quarentena) e avisos da verificação
  const alertas = achar(arvore, (n) => n.props?.role === 'alert');
  const aviso = {
    exibido: alertas.length > 0,
    titulo: alertas.length ? textoDe(achar(alertas[0], (n) => n.type === 'h4')[0]) : null,
    motivos: alertas.length ? achar(alertas[0], (n) => n.type === 'li').map((n) => textoDe(n)) : [],
  };
  const statusBlocos = achar(arvore, (n) => n.props?.role === 'status');
  const blocoAvisos = statusBlocos.find((n) => textoDe(achar(n, (m) => m.type === 'h4')[0]) === 'Avisos da verificação');
  const avisosDaVerificacao = {
    exibido: !!blocoAvisos,
    itens: blocoAvisos ? achar(blocoAvisos, (n) => n.type === 'li').map((n) => textoDe(n)) : [],
  };
  const blocoMsg = statusBlocos.find(
    (n) => achar(n, (m) => m.type === 'h4').length === 0 && achar(n, (m) => m.type === 'p').some((p) => textoDe(p) === (aiReview?.mensagemDaExtracao ?? '\u0000'))
  );
  const mensagem = { exibido: !!blocoMsg, texto: blocoMsg ? textoDe(achar(blocoMsg, (n) => n.type === 'p')[0]) : null, quantidadeDeBlocos: statusBlocos.filter((n) => achar(n, (m) => m.type === 'h4').length === 0 && achar(n, (m) => m.type === 'p').some((p) => textoDe(p) === (aiReview?.mensagemDaExtracao ?? '\u0000'))).length };
  const blocoSusp = statusBlocos.find((n) => textoDe(achar(n, (m) => m.type === 'h4')[0]).startsWith('📊'));
  const cartaoDeSuspensao = {
    exibido: !!blocoSusp,
    titulo: blocoSusp ? textoDe(achar(blocoSusp, (n) => n.type === 'h4')[0]) : null,
    motivo: blocoSusp ? textoDe(achar(blocoSusp, (n) => n.type === 'p')[0]) : null,
  };

  // 3. CORPO
  const recipiente = achar(arvore, (n) => classe(n).includes('whitespace-pre-wrap'));
  const caixaCorpo = achar(arvore, (n) => classe(n).includes('rounded-lg border p-6'));
  const linhasExibidas = recipiente.length ? filhos(recipiente[0]).map((l) => textoDe(l)) : [];
  const textoExibido = linhasExibidas.join('\n');
  const corpo = {
    exibido: recipiente.length > 0,
    estiloDeQuarentena: caixaCorpo.length ? classe(caixaCorpo[0]).includes('border-amber-400') : null,
    linhasExibidas: linhasExibidas.length,
    mesmoConteudoQueOTextoRecebido: recipiente.length > 0 && normalizar(textoExibido) === normalizar(aiReview?.review ?? ''),
    padroesDoExtratorNoCorpoExibido: PADROES.map((p, i) => (p.pattern.test(textoExibido) ? i : -1)).filter((i) => i >= 0),
  };

  // 4. CÓPIA (manipulador real, writeText simulado)
  const copia = await acionarCopia(arvore, setCopied);
  const copiado = {
    ...copia,
    argumentoIgualAoTextoRecebidoIntegral: copia.argumento !== null && copia.argumento === aiReview?.review,
    argumentoComprimento: copia.argumento === null ? null : copia.argumento.length,
    argumentoSha256: copia.argumento === null ? null : sha256(copia.argumento),
  };

  // Conferência cruzada árvore x HTML
  const htmlTem = (s: string) => html.includes(s);
  const confere = {
    destaque: htmlTem('Nota Científica') === destaque.exibido,
    aviso: htmlTem('role="alert"') === aviso.exibido,
    avisosDaVerificacao: htmlTem('Avisos da verificação') === avisosDaVerificacao.exibido,
    cartaoDeSuspensao: htmlTem('📊 ') === cartaoDeSuspensao.exibido,
    corpo: htmlTem('whitespace-pre-wrap') === corpo.exibido,
    botaoDeCopia: htmlTem('Copiar parecer em markdown') === copiado.botaoExibido,
    mensagem: htmlTem('Veredito não identificado no texto do parecer simulado.') === mensagem.exibido,
  };
  // A ORDEM no markup (posição de cada superfície presente no HTML renderizado)
  const pos: Array<[string, number]> = [
    ['aviso-A27', html.indexOf('role="alert"')],
    ['avisos-da-verificacao', html.indexOf('Avisos da verificação')],
    ['cartao-de-suspensao', html.indexOf('📊 Destaque')],
    ['mensagem', html.indexOf('Veredito não identificado no texto do parecer simulado.')],
    ['destaque', html.indexOf('Nota Científica')],
    ['corpo', html.indexOf('whitespace-pre-wrap')],
  ];
  const ordem = pos.filter(([, p]) => p >= 0).sort((a, b) => a[1] - b[1]).map(([n]) => n);
  const ocorrenciasDaMensagemNoHtml = html.split('Veredito não identificado no texto do parecer simulado.').length - 1;
  return { destaque, aviso, avisosDaVerificacao, cartaoDeSuspensao, mensagem, ordemNoMarkup: ordem, ocorrenciasDaMensagemNoHtml, corpo, copiado, arvoreConfereComHtml: Object.values(confere).every(Boolean), confereDetalhe: confere, htmlBytes: html.length };
}

const logCaminho = (ls: string[]): string => {
  if (ls.some((l) => l.includes('Grade extraída da DECISÃO EDITORIAL'))) return 'editorial';
  if (ls.some((l) => l.includes('Grade extraída do texto completo (fallback)'))) return 'texto-completo';
  if (ls.some((l) => l.includes('Não foi possível extrair grade'))) return 'nenhum';
  // A guarda de 50 caracteres devolve `null` SEM log do extrator; a rota registra apenas o log do estado sem padrão.
  if (ls.some((l) => l.includes('Nenhum padrão de nota ou veredicto reconhecido'))) return 'nulo-sem-log (guarda de 50)';
  return 'nao-executado';
};
const logsDeNota = (ls: string[]) =>
  ls.filter((l) => /Grade extraída|Não foi possível extrair grade|Nota IA|Divergência|Fallback para nota automática|Nenhum padrão de nota ou veredicto reconhecido|Nota suspensa|Classificação SUSPENSA|Resultado:|Score final/.test(l));

test('SONDA DEPOIS: modo texto fixado (matriz, controles construídos e adicionais) e modo captura (contextos)', async () => {
  fs.mkdirSync(SAIDA, { recursive: true });
  fs.mkdirSync(path.join(SAIDA, 'contextos'), { recursive: true });

  // ⚠ CONTROLE de rede: `fetch` simulado, que LANÇA e conta. Nenhuma célula pode tocá-lo (`USE_RAG_SEMANTIC` desligado).
  const espiaoFetch = jest.spyOn(globalThis as any, 'fetch').mockImplementation((() => { throw new Error('SONDA: fetch proibido'); }) as any);

  // ---- sanidade das condições, pelas funções REAIS (módulo real, fora do mock)
  const sanidade = CONDICOES.map((c) => {
    const raw = c.fazer();
    const av = real.classificarAvaliacaoRecebida(raw);
    const co = real.avaliarCoerencia(raw);
    const el = real.elegivelParaClassificacao(av, c.coerenciaNula ? null : co);
    return { id: c.id, avaliacao: av.estado, coerencia: co.conclusao, elegivel: el.elegivel, causa: el.causa, rotulo: el.rotulo, motivo: el.motivo };
  });

  // ================= MODO TEXTO FIXADO =================
  const celulas: any[] = [];
  for (const ap of APRESENTACOES) {
    for (const cond of CONDICOES) {
      if (ap.aplicaA !== 'todas' && ap.aplicaA !== cond.id) continue;
      for (const t of TEXTOS) {
        const texto = t.texto + ap.sufixo;
        const rep = replica(texto);
        const exec = await executar(ap.ajustar(cond.fazer()), 'texto-fixado', texto, cond.coerenciaNula);
        const c = exec.corpo;
        const aiReview = aiReviewDaPagina(c);
        const sup = c.success && c.review ? await superficies(aiReview) : null;
        const caminho = logCaminho(exec.logs);
        const eds = c.metadata?.estadoDaExtracao;
        const suspensa = c.notaSuspensa !== null && c.notaSuspensa !== undefined;
        // Suspensa: a extração NÃO corre (R1), e o estado é `nao_executada_por_suspensao`, sem nota, sem veredicto e sem mensagem.
        // Não suspensa: a réplica prevê o caminho; com padrão, o estado é `padrao_reconhecido` e a mensagem é nula; sem padrão,
        // o estado é `nenhum_padrao_reconhecido`, nota e veredicto são NULOS (sem valor automático) e a mensagem é a literal do pedido.
        const confereReplica = suspensa
          ? caminho === 'nao-executado' && eds === 'nao_executada_por_suspensao' && c.mensagemDaExtracao === null && c.nota == null && c.veredicto == null
          : (rep.caminho === caminho || (rep.caminho === 'curto' && ['nenhum', 'nulo-sem-log (guarda de 50)'].includes(caminho))) &&
            (rep.nota !== null
              ? eds === 'padrao_reconhecido' && c.nota === rep.nota && c.veredicto === rep.veredicto && c.mensagemDaExtracao === null
              : eds === 'nenhum_padrao_reconhecido' && c.nota == null && c.veredicto == null && c.mensagemDaExtracao === MENSAGEM);
        celulas.push({
          apresentacao: ap.id,
          condicao: cond.id,
          condicaoConstruida: cond.construida,
          texto: t.id,
          textoExtensao: texto.length,
          chamadasAoCliente: exec.chamadas,
          status: exec.status,
          resposta: {
            success: c.success,
            nota: c.nota ?? null,
            veredicto: c.veredicto ?? null,
            ...camposDaResposta(c),
            notaSuspensa: c.notaSuspensa ? { suspensa: c.notaSuspensa.suspensa, causa: c.notaSuspensa.causa, rotulo: c.notaSuspensa.rotulo, motivo: c.notaSuspensa.motivo } : null,
            validationEstado: c.validation?.estado ?? null,
            validationIssues: c.validation?.issues ?? null,
            validationInconclusivos: c.validation?.inconclusivos ?? null,
            validationWarnings: c.validation?.warnings ?? null,
            reviewIgualAoTextoFixado: c.review === texto,
          },
          extracao: { replica: rep, caminhoObservadoNoLog: caminho, confereReplicaComTratador: confereReplica },
          logsDeNota: logsDeNota(exec.logs),
          superficies: sup,
        });
      }
    }
  }

  // ---- estado de A.27 `nao_confirmado`: CONSTRUÍDO a partir de uma resposta REAL sem padrão reconhecido (T2a e T2b, nas
  //      duas condições elegíveis). ⚠ A construção é DECLARADA e acontece SÓ no instrumento: o tratador real SEMPRE devolve
  //      `validation`, e a produção NÃO foi alterada para tornar esse estado alcançável.
  const naoConfirmadoConstruido: any[] = [];
  for (const cond of CONDICOES.filter((c) => c.id.startsWith('elegivel'))) {
    for (const t of TEXTOS.filter((x) => ['T2a-ausencia-sem-secao', 'T2b-ausencia-secao-sem-padrao'].includes(x.id))) {
      const exec = await executar(cond.fazer(), 'texto-fixado', t.texto, false);
      const c = exec.corpo;
      const real0 = aiReviewDaPagina(c);
      for (const variante of ['validation-retirada', 'validation-invalidada'] as const) {
        const aiReview = variante === 'validation-retirada' ? { ...real0, validation: undefined } : { ...real0, validation: { ...(c.validation as object), version: 'versao-invalida-do-instrumento' } };
        naoConfirmadoConstruido.push({
          condicao: cond.id,
          texto: t.id,
          variante,
          procedencia: 'CONSTRUÍDO: resposta real do tratador, sem padrão reconhecido, com validation retirada ou invalidada SÓ no instrumento',
          respostaReal: { nota: c.nota ?? null, veredicto: c.veredicto ?? null, ...camposDaResposta(c), validationEstadoDaRespostaReal: c.validation?.estado ?? null },
          superficies: await superficies(aiReview),
        });
      }
    }
  }

  // ---- controles adicionais (modo texto fixado), nas duas condições elegíveis, apresentação aprovada
  const celulasAdicionais: any[] = [];
  for (const condAd of CONDICOES.filter((c) => c.id.startsWith('elegivel'))) {
    for (const t of ADICIONAIS) {
      const rep = replica(t.texto);
      const exec = await executar(condAd.fazer(), 'texto-fixado', t.texto);
      const c = exec.corpo;
      const aiReview = aiReviewDaPagina(c);
      const sup = c.success && c.review ? await superficies(aiReview) : null;
      celulasAdicionais.push({
        condicao: condAd.id,
        texto: t.id,
        descricao: t.descricao,
        leituraHumana: t.leituraHumana,
        textoIntegral: t.texto,
        anatomia: anatomia(t.texto),
        resposta: { success: c.success, nota: c.nota ?? null, veredicto: c.veredicto ?? null, ...camposDaResposta(c), validationEstado: c.validation?.estado ?? null },
        extracao: { replica: rep, caminhoObservadoNoLog: logCaminho(exec.logs) },
        superficies: sup
          ? { destaque: sup.destaque, aviso: sup.aviso, mensagem: sup.mensagem, ordemNoMarkup: sup.ordemNoMarkup, corpo: { exibido: sup.corpo.exibido, estiloDeQuarentena: sup.corpo.estiloDeQuarentena }, copiadoIgualAoTexto: sup.copiado.argumentoIgualAoTextoRecebidoIntegral }
          : null,
      });
    }
  }

  // ================= MODO CAPTURA =================
  const capturas: any[] = [];
  const textoQualquer = TEXTOS[0].texto;
  for (const cond of CONDICOES) {
    const exec = await executar(cond.fazer(), 'captura', textoQualquer, cond.coerenciaNula);
    fs.writeFileSync(path.join(SAIDA, 'contextos', `ctx-${cond.id}.txt`), exec.contexto);
    const linhas = exec.contexto.split('\n');
    const nLinha = (i: number) => i + 1;
    const alvos = [
      'Referência Automatizada',
      'Pontuação automática',
      'Sugestão automática',
      'referência automática',
      'Nota não calculada',
      'Destaque de nota e veredito suspenso',
      'classificação global está SUSPENSA',
      'CLASSIFICAÇÃO SUSPENSA',
      'Score = ',
      DESCRICOES_DO_PEDIDO[0],
      DESCRICOES_DO_PEDIDO[1],
      DESCRICOES_DO_PEDIDO[2],
      DESCRICOES_DO_PEDIDO[3],
    ];
    const ocorrencias: Record<string, Array<{ linha: number; texto: string }>> = {};
    for (const a of alvos) {
      ocorrencias[a] = linhas.map((l, i) => (l.includes(a) ? { linha: nLinha(i), texto: l } : null)).filter(Boolean) as any;
    }
    const motivo = sanidade.find((s) => s.id === cond.id)!.motivo as string | null;
    const linhasDoMotivo = motivo ? linhas.map((l, i) => (l.includes(motivo) ? nLinha(i) : null)).filter((x) => x !== null) : [];
    capturas.push({
      condicao: cond.id,
      construida: cond.construida,
      modo: 'captura',
      statusDaResposta: exec.status,
      respostaDeInterrupcao: { error: exec.corpo?.error ?? null, details: exec.corpo?.details ?? null },
      chamadasAoCliente: exec.chamadas,
      sha256Contexto: sha256(exec.contexto),
      caracteresContexto: exec.contexto.length,
      bytesContexto: Buffer.byteLength(exec.contexto, 'utf8'),
      linhasContexto: linhas.length,
      sha256Requisicao: sha256(JSON.stringify(cond.fazer())),
      coerenciaForcadaNula: cond.coerenciaNula,
      sha256System: sha256(exec.system),
      bytesSystem: Buffer.byteLength(exec.system, 'utf8'),
      motivoEsperado: motivo,
      motivoOcorrenciasNoContexto: motivo ? exec.contexto.split(motivo).length - 1 : null,
      motivoLinhasNoContexto: linhasDoMotivo,
      ocorrencias,
    });
  }

  fs.writeFileSync(
    path.join(SAIDA, 'sonda.json'),
    JSON.stringify(
      {
        gerado: 'sonda temporária da Fase 2, DEPOIS; ver instrumentos/',
        ambiente: { node: process.version, plataforma: process.platform + '/' + process.arch },
        chamadasAFetch: espiaoFetch.mock.calls.length,
        chamadasAoClienteDoModeloNoTotal: estado.chamadas,
        sanidadeDasCondicoes: sanidade,
        textos: TEXTOS.map((t) => ({ id: t.id, descricao: t.descricao, leituraHumana: t.leituraHumana, texto: t.texto, anatomia: anatomia(t.texto), previsaoDaReplica: replica(t.texto) })),
        condicoes: CONDICOES.map((c) => ({ id: c.id, descricao: c.descricao, construida: c.construida })),
        apresentacoes: APRESENTACOES.map((a) => ({ id: a.id, sufixoAcrescentado: a.sufixo, aplicaA: a.aplicaA })),
        celulas,
        naoConfirmadoConstruido,
        capturas,
        celulasAdicionais,
        requisicoes: Object.fromEntries(CONDICOES.map((c) => [c.id, c.fazer()])),
      },
      null,
      1
    )
  );
  expect(celulas.length).toBeGreaterThan(0);
  expect(espiaoFetch.mock.calls.length).toBe(0);
  espiaoFetch.mockRestore();
});
