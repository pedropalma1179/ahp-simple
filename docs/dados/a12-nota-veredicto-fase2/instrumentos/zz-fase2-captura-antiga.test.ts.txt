const respondente = (i: number, cr: number | null, status?: string) => ({
  id: `resp-${i}`,
  name: `Respondente ${i}`,
  ...(cr === null ? {} : { cr, metrics: { avgCR: cr } }),
  ...(status === undefined
    ? { status: cr === null ? 'DESCONHECIDO' : cr <= 0.1 ? 'CONFIÁVEL' : cr <= 0.2 ? 'SUSPEITO' : 'CRÍTICO' }
    : { status }),
});

type Execucao = { corpo: any; status: number; chamadas: number; contexto: string; system: string; logs: string[] };

async function executar(payload: any, modo: 'captura' | 'texto-fixado', texto: string, coerenciaNula = false): Promise<Execucao> {
  estado.modo = modo;
  estado.texto = texto;
  estado.contexto = '';
  estado.system = '';
  estado.coerenciaNula = coerenciaNula;
  logs = [];
  const antes = estado.chamadas;
  const res = await POST({ json: async () => JSON.parse(JSON.stringify(payload)) } as any);
  const corpo = await res.json();
  estado.coerenciaNula = false;
  return { corpo, status: res.status ?? 200, chamadas: estado.chamadas - antes, contexto: estado.contexto, system: estado.system, logs: [...logs] };
}

const SAIDA = process.env.FASE2_SAIDA || path.join(process.cwd(), 'zz-fase2-saida');

test('SONDA FASE 2: captura dos sete contextos no código ANTERIOR, e identidade com a Fase 1', async () => {
  fs.mkdirSync(path.join(SAIDA, 'contextos-antigos'), { recursive: true });
  const fase1 = JSON.parse(fs.readFileSync(path.join(process.cwd(), 'docs/dados/a12-nota-veredicto-fase1/captura.json'), 'utf8'));
  const tabela: Record<string, any> = {};
  for (const c of (fase1.capturas || fase1)) tabela[c.condicao] = c;
  const espiaoFetch = jest.spyOn(globalThis as any, 'fetch').mockImplementation((() => { throw new Error('SONDA: fetch proibido'); }) as any);
  const saida: any[] = [];
  for (const cond of CONDICOES) {
    const exec = await executar(cond.fazer(), 'captura', 'texto qualquer', cond.coerenciaNula);
    fs.writeFileSync(path.join(SAIDA, 'contextos-antigos', `ctx-${cond.id}.txt`), exec.contexto);
    const ref = tabela[cond.id];
    saida.push({
      condicao: cond.id,
      construida: cond.construida,
      sha256Contexto: sha256(exec.contexto),
      bytesContexto: Buffer.byteLength(exec.contexto, 'utf8'),
      linhasContexto: exec.contexto.split('\n').length,
      sha256System: sha256(exec.system),
      sha256RequisicaoReexecucao: sha256(JSON.stringify(cond.fazer())),
      fase1: ref ? { sha256Contexto: ref.sha256Contexto, bytesContexto: ref.bytesContexto, sha256System: ref.sha256System } : null,
      identicoAFase1: !!ref && ref.sha256Contexto === sha256(exec.contexto) && ref.sha256System === sha256(exec.system) && ref.bytesContexto === Buffer.byteLength(exec.contexto, 'utf8'),
    });
  }
  fs.writeFileSync(path.join(SAIDA, 'captura-antiga.json'), JSON.stringify({ chamadasAFetch: espiaoFetch.mock.calls.length, chamadasAoCliente: estado.chamadas, capturas: saida }, null, 1));
  expect(espiaoFetch.mock.calls.length).toBe(0);
  espiaoFetch.mockRestore();
  expect(saida.every((s) => s.identicoAFase1)).toBe(true);
});
